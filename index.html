const match1 = url.match(/\/d\/([a-zA-Z0-9_-]+)/); const match2 = url.match(/id=([a-zA-Z0-9_-]+)/); if (match1 && match1[1]) { fileId = match1[1]; } else if (match2 && match2[1]) { fileId = match2[1]; } if (fileId) { // Direct reliable high-res Google Content URL that bypasses CORS & iframe blocking return `https://lh3.googleusercontent.com/d/${fileId}`; } return url; } function updateAllLogoImages(rawUrl) { const finalUrl = convertGoogleDriveUrl(rawUrl); updateAllLogoImages(directUrl); localStorage.setItem('skdb_custom_logo_url', customUrl); document.getElementById('modalLogoSettings').classList.add('hidden'); showToast('Logo Google Drive berjaya disambungkan!'); } else { showToast('Sila masukkan pautan Google Drive logo anda.', 'error'); } }); document.getElementById('btnResetLogo')?.addEventListener('click', () => { localStorage.removeItem('skdb_custom_logo_url');
